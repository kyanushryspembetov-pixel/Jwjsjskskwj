local function corner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 10)
    c.Parent = parent
    return c
end

local function stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or C.line
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = parent
    return s
end

local function tween(obj, duration, props)
    local t = TweenService:Create(obj, TweenInfo.new(duration, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props)
    t:Play()
    return t
end

local function label(parent, text, size, color, font)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Text = text
    l.TextColor3 = color or C.white
    l.TextSize = size or 14
    l.Font = font or Enum.Font.Gotham
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.TextYAlignment = Enum.TextYAlignment.Center
    l.Parent = parent
    return l
end

local function addToggle(titleText, description, initial, callback)
    local enabled = initial == true
    local f = card(62)
    local a = label(f, titleText, 12, C.white, Enum.Font.GothamSemibold)
    a.Position = UDim2.fromOffset(13, 7)
    a.Size = UDim2.new(1, -105, 0, 20)
    local b = label(f, description or "", 10, C.muted)
    b.Position = UDim2.fromOffset(13, 29)
    b.Size = UDim2.new(1, -105, 0, 18)
    local button = Instance.new("TextButton")
    button.Size = UDim2.fromOffset(62, 28)
    button.AnchorPoint = Vector2.new(1, 0.5)
    button.Position = UDim2.new(1, -12, 0.5, 0)
    button.BackgroundColor3 = enabled and C.white or C.off
    button.Text = enabled and "ON" or "OFF"
    button.TextColor3 = enabled and C.bg or C.white
    button.TextSize = 10
    button.Font = Enum.Font.GothamBold
    button.AutoButtonColor = false
    button.Parent = f
    corner(button, 8)
    local function refresh()
        tween(button, 0.16, {BackgroundColor3 = enabled and C.white or C.off})
        button.Text = enabled and "ON" or "OFF"
        button.TextColor3 = enabled and C.bg or C.white
    end
    button.Activated:Connect(function()
        enabled = not enabled
        refresh()
        local ok, err = pcall(callback, enabled)
        if not ok then warn("[AVASA HUB] " .. tostring(err)) end
    end)
end

local function makePage(name, description, build)
    local page = {name = name, description = description, build = build}
    table.insert(pages, page)
    local nav = Instance.new("TextButton")
    nav.Name = name:gsub("%s+", "") .. "Button"
    nav.Size = UDim2.new(1, 0, 0, 36)
    nav.BackgroundColor3 = C.panel
    nav.Text = "  " .. name
    nav.TextColor3 = C.muted
    nav.TextSize = 11
    nav.Font = Enum.Font.GothamSemibold
    nav.TextXAlignment = Enum.TextXAlignment.Left
    nav.AutoButtonColor = false
    nav.Parent = navHolder
    corner(nav, 8)
    navButtons[page] = nav
    nav.MouseEnter:Connect(function()
        if activePage ~= page then tween(nav, 0.12, {BackgroundColor3 = C.panel2, TextColor3 = C.white}) end
    end)
    nav.MouseLeave:Connect(function()
        if activePage ~= page then tween(nav, 0.12, {BackgroundColor3 = C.panel, TextColor3 = C.muted}) end
    end)
    nav.Activated:Connect(function()
        if activePage == page then return end
        activePage = page
        for p, btn in pairs(navButtons) do
            tween(btn, 0.14, {
                BackgroundColor3 = p == page and C.white or C.panel,
                TextColor3 = p == page and C.bg or C.muted
            })
        end
        pageHeader.Text = page.name
        pageSub.Text = page.description
        clearPage()
        page.build()
        pageArea.CanvasPosition = Vector2.zero
        pageArea.Position = UDim2.fromOffset(18, 69)
        pageArea.BackgroundTransparency = 1
        tween(pageArea, 0.18, {Position = UDim2.fromOffset(12, 69)})
    end)
    return page
end

makePage("Misc", "Small interface utilities", function()
    addToggle("Show FPS in status", "Update the top status label", false, function(on)
        if fpsConnection then
            fpsConnection:Disconnect()
            fpsConnection = nil
        end
        if on then
            local last = os.clock()
            local frames = 0
            fpsConnection = RunService.RenderStepped:Connect(function()
                frames += 1
                local now = os.clock()
                if now - last >= 1 then
                    if not gui.Parent or status.Parent == nil then
                        if fpsConnection then fpsConnection:Disconnect(); fpsConnection = nil end
                        return
                    end
                    status.Text = "●  " .. tostring(frames) .. " FPS"
                    frames = 0
                    last = now
                end
            end)
        else
            status.Text = "●  READY"
        end
    end)
    addButton("Close hub", "You can reopen it with the small AVASA button", function()
        tween(window, 0.18, {Position = UDim2.new(0.5, 0, 0.55, 20)})
        task.wait(0.16)
        window.Visible = false
        reopen.Visible = true
    end)
end)

-- Smooth drag for mouse and touch.
do
    local dragging = false
    local dragInput, dragStart, startPos
    top.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = window.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    top.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            window.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end
